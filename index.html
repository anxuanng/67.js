Assembled

A scene in a few lines

import { Renderer67 } from '67js';
import { createModel } from '67js/addons/model';

const renderer = new Renderer67(canvas, createModel(), {
  samples: 4,
  memoryBudgetMiB: 256,
});

renderer.render({
  eye: [4, 3, 7],
  target: [0, 1.5, 0],
  assemblyAmount: 0.5,
});
